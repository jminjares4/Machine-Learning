\section*{2}

\subsection*{(a)}

E-step:
\[
p=\frac{e^{-1/2}}{e^{-1/2}+e^{-9/2}}
=\frac{1}{1+e^{-4}}\approx0.98.
\]
By symmetry, the other component has the same probability for $x=1$.

M-step:
\begin{align*}
\mu_1
&=\frac{p(-1)+(1-p)(1)}{p+(1-p)}
=1-2p\approx-0.96,\\
\mu_2
&=\frac{(1-p)(-1)+p(1)}{(1-p)+p}
=2p-1\approx0.96.
\end{align*}
\[
\boxed{\mu_1\approx-0.96,\qquad\mu_2\approx0.96}
\]

\subsection*{(b)}

No. By symmetry, let $(\mu_1,\mu_2)=(-a,a)$ with $a>0$.
\[
p=\frac{1}{1+e^{-2a}},
\qquad
a_{\text{new}}=2p-1
=\frac{e^{2a}-1}{e^{2a}+1}
=\tanh(a).
\]
Since $0<\tanh(a)<a$, the sequence decreases to $L$ satisfying $L=\tanh(L)$. Thus, $L=0$.
\[
\boxed{(\mu_1,\mu_2)\longrightarrow(0,0)}
\]

\subsection*{(c)}

Since $\mu_1=\mu_2=2$, both components have equal probabilities.
\[
\mu_1=\mu_2
=\frac{\frac12(-1)+\frac12(1)}
{\frac12+\frac12}=0.
\]
The means remain at zero.
\[
\boxed{(\mu_1,\mu_2)=(0,0)}
\]

\subsection*{(d)}

Each point is assigned to its nearest center:
\[
-1\rightarrow\mu_1,\qquad
1\rightarrow\mu_2.
\]
The updated centers are
\[
\mu_1=-1,\qquad\mu_2=1.
\]
The assignments remain unchanged.
\[
\boxed{(\mu_1,\mu_2)=(-1,1)}
\]
